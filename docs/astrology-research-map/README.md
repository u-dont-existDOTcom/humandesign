flowchart TD
    IN["Source: dated book / codex / living daykeeper / method"]

    RESULT["Report separately by tradition, outcome class and cohort"]
    IN --> A --> B

    HYP --> READY
    READY -- "Unknown/conflicting" --> HOLD
    READY -- "Eligible" --> FREEZE --> HUMAN --> EVAL --> RESULT
